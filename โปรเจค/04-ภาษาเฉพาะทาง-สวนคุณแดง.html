Khun Daeng Garden — Tree Booking Web Application

ภาษาหลัก: TypeScript / TSX
Framework: Next.js 14 (App Router)
CSS: CSS Modules + Global CSS
ฐานข้อมูล: PostgreSQL (Prisma ORM)
Hosting: Vercel
1. คำศัพท์ภาษาเฉพาะทาง (Domain-Specific Terminology)
คำศัพท์ ภาษาอังกฤษ ความหมายในโปรเจค
เว็บแอปพลิเคชัน Web Application ระบบซอฟต์แวร์ที่ทำงานผ่านเบราว์เซอร์อินเทอร์เน็ต สร้างด้วย Next.js 14
คอมโพเนนต์ Component ชิ้นส่วนย่อยของ UI ที่สามารถนำกลับมาใช้ซ้ำได้ เช่น Navbar, Footer, Button, Card
เอพีไอ API (Application Programming Interface) ชุดคำสั่งสำหรับรับ-ส่งข้อมูลระหว่าง Client กับ Server อยู่ในโฟลเดอร์ src/app/api/
ฐานข้อมูล Database (PostgreSQL + Prisma ORM) ระบบจัดเก็บข้อมูลต้นไม้ ผู้ใช้ และการจอง ใช้ Prisma เป็นตัวกลางสื่อสารกับ PostgreSQL
การจอง Booking ข้อมูลการสั่งจองต้นไม้ล่วงหน้า มีสถานะ: PENDING, CONFIRMED, COMPLETED, CANCELLED
มัดจำ Deposit เงินจ่ายล่วงหน้าเพื่อยืนยันการจอง ลูกค้าอัปโหลดสลิปผ่าน PaymentModal
สต็อก Stock จำนวนต้นไม้คงเหลือในระบบ เก็บในฐานข้อมูล field stock ของ model Tree
โปรโมชั่น Promotion ต้นไม้ที่ลดราคาพิเศษ กำหนดด้วย field isPromotion: true และ promotionEndDate
แอดมิน Admin บทบาทผู้ดูแลระบบ มีสิทธิ์จัดการสินค้า ดูรายงาน และแก้ไขการตั้งค่า user.role === "admin"
สลิป Payment Slip รูปภาพหลักฐานการชำระเงิน ที่ลูกค้าอัปโหลดผ่านระบบเพื่อยืนยันมัดจำ
CSS Variables (ตัวแปร CSS) CSS Custom Properties ตัวแปรสีและรูปแบบที่ใช้ร่วมกันทั้งระบบ เช่น --primary, --secondary, --radius
กลาสมอร์ฟิซึม Glassmorphism เอฟเฟกต์กระจกฝ้าของ Navbar ใช้ backdrop-filter: blur(16px) และ class .glass
เลย์เอาต์ Layout โครงสร้างหน้าหลักของแอปใน src/app/layout.tsx ครอบคลุม Navbar, เนื้อหา, Footer
รูทเลย์เอาต์ Root Layout ไฟล์ layout.tsx ที่ Next.js ใช้เป็นแม่แบบให้กับทุกหน้า
อินไลน์เอดิต Inline Edit คอมโพเนนต์ InlineEdit ที่ให้ Admin แก้ไขข้อความ/สีบนหน้าเว็บได้โดยตรง โดยบันทึกลงฐานข้อมูล SiteSetting
ฟอนต์ Prompt Prompt Font (Google Fonts) ฟอนต์ภาษาไทยหลักของโปรเจค โหลดผ่าน next/font/google กำหนดเป็น --font-prompt

interface ButtonProps extends React.ButtonHTMLAttributes<HTMLButtonElement> {
  variant?: "primary" | "secondary" | "outline" | "ghost";
  size?: "sm" | "md" | "lg";
  fullWidth?: boolean;
}

export const Button = React.forwardRef<HTMLButtonElement, ButtonProps>(
  ({ variant = "primary", size = "md", fullWidth = false, children, ...props }, ref) => {
    const classes = [
      styles.button,
      styles[variant],
      styles[size],
      fullWidth ? styles.fullWidth : "",
    ].filter(Boolean).join(" ");

    return <button ref={ref} className={classes} {...props}>{children}</button>;
  }
);

Navbar มี 3 ระดับ: Top Bar (โปรโมชั่น/ติดต่อ), Middle Bar (โลโก้/ค้นหา/ไอคอน) และ Bottom Bar (เมนูหลัก) รองรับ Mobile Bottom Navigation

"use client";  // Client Component — ใช้ useState, useRouter

import { useCart } from "@/lib/CartContext";    // Context จำนวนสินค้าในตะกร้า

export function Navbar({ topBarText = "ฟรีปุ๋ยหมักเมื่อสั่งซื้อเกิน 1,000 บาท" }) {

  return (
    <>

        {/* 1. Top Bar — แถบบนแสดงข้อความโปรโมชั่น */}

7. Homepage — Hero Section & Best Sellers (page.tsx)
ที่มาของโค้ด: src/app/page.tsx — ดึงข้อมูลจากฐานข้อมูลด้วย Prisma และแสดงผลบนหน้าหลัก
ใช้ Pattern: Server Component + Prisma ORM + Fallback ด้วย Mock Data
src/app/page.tsx — ดึงข้อมูลสินค้าขายดี TSX 
export const dynamic = "force-dynamic";  // โหลดข้อมูลใหม่ทุกครั้ง

export default async function Home() {
  try {
    // ดึง Best Sellers — จาก BookingItem ที่ status = COMPLETED
    const bestSellingData = await prisma.bookingItem.groupBy({
      by: ["treeId"],
      where: { booking: { status: "COMPLETED" } },
      _sum: { quantity: true },
      orderBy: { _sum: { quantity: "desc" } },
      take: 5,
    });

    // ดึง Promotional Trees — ต้นไม้โปรโมชั่น
    promotionalTrees = await prisma.tree.findMany({
      where: {
        isPromotion: true,
        OR: [
          { promotionEndDate: null },
          { promotionEndDate: { gte: new Date() } },  // ยังไม่หมดอายุ
        ],
      },
      take: 4,
    });

    // ดึง Site Settings — ข้อความ/สีจากฐานข้อมูล
    const settings = await prisma.siteSetting.findMany();
    settingsMap = settings.reduce((acc, curr) => {
      acc[curr.key] = curr.value;
      return acc;
    }, {} as Record<string, string>);

  } catch (error) {
    // Fallback — ใช้ Mock Data เมื่อฐานข้อมูลไม่พร้อม
    featuredTrees = MOCK_TREES.slice(0, 4);
    heroTree = MOCK_TREES[0];
    settingsMap = MOCK_SITE_SETTINGS;
  }

  // ดึง hero_title จาก SiteSetting (ปรับได้โดย Admin)
  const heroTitle = settingsMap["hero_title"] || "สวนสวยเริ่มต้นที่ สวนคุณแดง";

  return (
    <main>
      {/* Hero Section */}
      <section style={{ backgroundColor: "#fefcf9", padding: "4rem 0 6rem" }}>
        <ScrollAnimation animation="fade-up">
          <h1>{heroTitle}</h1>
          <Link href="/shop">
            <Button size="lg" style={{ borderRadius: "50px" }}>
              จองต้นไม้ →
            </Button>
          </Link>
        </ScrollAnimation>
        
        {/* Parallax Image ของต้นไม้ Hero */}
        <ParallaxSection speed={0.3}>
          ...
        </ParallaxSection>
      </section>

      {/* Best Sellers Slider — ImageSlider Component */}
      <ImageSlider trees={weeklyBestSellingTreesWithCount}
        title="ขายดีสัปดาห์นี้"
        subtitle="Weekly Best Sellers"
      />
    </main>
  );
}
